DROP TABLE IF EXISTS `participantes_funcoes`;
DROP TABLE IF EXISTS `funcoes`;
DROP TABLE IF EXISTS `escalas`;
DROP TABLE IF EXISTS `new_table`;
DROP TABLE IF EXISTS `users`;

CREATE TABLE `users` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `username` varchar(50) NOT NULL,
  `email` varchar(100) NOT NULL,
  `password` varchar(255) NOT NULL,
  `is_admin` tinyint(1) NOT NULL DEFAULT 0,
  PRIMARY KEY (`id`),
  UNIQUE KEY `username` (`username`),
  UNIQUE KEY `email` (`email`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

CREATE TABLE `new_table` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `NOME` varchar(45) NOT NULL,
  `EMAIL` varchar(140) NOT NULL,
  `TELEFONE` varchar(20) NOT NULL,
  `NASCIMENTO` date DEFAULT NULL,
  `ENDERECO` varchar(250) DEFAULT NULL,
  `tipo` varchar(20) DEFAULT NULL,
  PRIMARY KEY (`id`),
  UNIQUE KEY `EMAIL` (`EMAIL`),
  CONSTRAINT `fk_new_table_users` FOREIGN KEY (`EMAIL`) REFERENCES `users` (`email`) ON DELETE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

CREATE TABLE `escalas` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `data_missa` date NOT NULL,
  `horario` time NOT NULL,
  `descricao` varchar(255) NOT NULL,
  PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

CREATE TABLE `funcoes` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `id_escala` int(11) NOT NULL,
  `funcao` varchar(100) NOT NULL,
  PRIMARY KEY (`id`),
  KEY `fk_funcoes_escala` (`id_escala`),
  CONSTRAINT `fk_funcoes_escala` FOREIGN KEY (`id_escala`) REFERENCES `escalas` (`id`) ON DELETE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

CREATE TABLE `participantes_funcoes` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `id_funcao` int(11) NOT NULL,
  `nome_participante` varchar(255) NOT NULL,
  PRIMARY KEY (`id`),
  KEY `fk_participantes_funcoes_funcao` (`id_funcao`),
  CONSTRAINT `fk_participantes_funcoes_funcao` FOREIGN KEY (`id_funcao`) REFERENCES `funcoes` (`id`) ON DELETE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;
